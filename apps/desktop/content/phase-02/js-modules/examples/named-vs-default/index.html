Three "files" in one demo — named + default exports

// math.js — named exports

export const TAX_RATE = 0.18;
export function withTax(p) {
  return p * (1 + TAX_RATE);
}

// theme.js — default export

const theme = {
  bg:   "#0a0a0d",
  fg:   "#ededee",
  accent: "#7fff5b",
};
export default theme;

// main.js — consumes both

import { TAX_RATE, withTax } from "./math.js";
import theme from "./theme.js";

const subtotal = 50000;
const total = withTax(subtotal);
console.log("TAX_RATE", TAX_RATE);
console.log("total",   total);
console.log("theme",   theme.accent);
