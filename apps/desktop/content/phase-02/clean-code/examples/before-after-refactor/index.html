✓ Clean — named constants, helpers, 6-line wrapper

const LOYALTY_DISCOUNT   = 0.10;
const BULK_THRESHOLD     = 100_000;
const BULK_DISCOUNT_RATE = 0.10;

const lineTotal = (item) => {
  const base = item.priceCents * item.qty;
  return item.isLoyalty
    ? Math.round(base * (1 - LOYALTY_DISCOUNT))
    : base;
};
const subtotal = (cart) => cart.reduce((s, i) => s + lineTotal(i), 0);
const bulkDiscount = (sub) => sub > BULK_THRESHOLD ? sub * BULK_DISCOUNT_RATE : 0;
const taxOf = (sub, ratePct) => Math.round(sub * ratePct / 100);

function checkout(cart, taxRatePct) {
  const sub  = subtotal(cart);
  const disc = bulkDiscount(sub);
  const tax  = taxOf(sub, taxRatePct);
  const final = Math.max(0, sub + tax - disc);
  return { subtotal: sub, discount: disc, tax, final };
}
