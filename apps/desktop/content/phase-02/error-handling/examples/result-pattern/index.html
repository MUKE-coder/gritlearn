Result<User> — error as data, not exception

type Result<T> =
  | { ok: true;  value: T }
  | { ok: false; error: string };

function lookupUser(id) {
  if (id === 0) return { ok: false, error: "Server unavailable" };
  if (id === 99) return { ok: false, error: "Not found" };
  return { ok: true, value: { id, name: "User " + id } };
}

function renderUser(r) {
  if (r.ok) return "→ " + r.value.name;
  return "✗ " + r.error;
}

Three calls:

lookupUser(1)
lookupUser(0)
lookupUser(99)

Compiler-equivalent: TypeScript forces you to check r.ok before touching r.value.

The pattern moves errors into the function's type signature — callers can't forget to handle them.
