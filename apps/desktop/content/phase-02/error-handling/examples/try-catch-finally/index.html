try / catch / finally — order of operations

function run(mode) {
  log("→ enter try");
  try {
    if (mode === "throw")    throw new Error("Math failed");
    if (mode === "throwFin") throw new Error("Cleanup test");
    log("✓ try succeeded");
    return "OK";
  } catch (err) {
    log("[catch] " + err.message);
    return "ERR";
  } finally {
    log("• finally — cleanup runs no matter what");
  }
}
