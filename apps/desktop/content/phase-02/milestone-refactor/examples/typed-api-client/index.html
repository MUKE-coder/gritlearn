The apiClient — what's in the wrapper

The fetch-wrapper shape every real frontend has. Validates with Zod (here, a tiny hand-rolled checker so the example needs no deps); throws typed errors the caller can branch with instanceof.

The shape

// lib/errors.ts — custom Error classes
export class NetworkError    extends Error { name = "NetworkError"; }
export class NotFoundError   extends Error { name = "NotFoundError"; }
export class ValidationError extends Error { name = "ValidationError"; }
export class ApiError        extends Error {
  constructor(public status: number, msg: string) { super(msg); }
}

// lib/apiClient.ts
export async function apiGet<T>(url: string, schema: Schema<T>): Promise<T> {
  let res: Response;
  try {
    res = await fetch(url);
  } catch {
    throw new NetworkError("Network unreachable");
  }
  if (res.status === 404) throw new NotFoundError(url);
  if (!res.ok)            throw new ApiError(res.status, await res.text());

  const json = await res.json();
  const parsed = schema.safeParse(json);
  if (!parsed.success) throw new ValidationError(parsed.error);
  return parsed.data;
}
