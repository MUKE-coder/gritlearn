"use client";
import { useState, useEffect } from "react";

export default function WeatherPage() {
  const [city, setCity] = useState("");
  const [data, setData] = useState<any>(null);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);

  const search = async () => {
    setLoading(true);
    setError(null);
    try {
      const res = await fetch(`https://geocoding-api.open-meteo.com/v1/search?name=${city}`);
      const json = await res.json();
      // no validation, no error class, json could be anything
      if (!json.results || json.results.length === 0) {
        setError("City not found");
        setLoading(false);
        return;
      }
      const hit = json.results[0];
      const fcRes = await fetch(`https://api.open-meteo.com/v1/forecast?latitude=${hit.latitude}&longitude=${hit.longitude}¤t=temperature_2m,weather_code,wind_speed_10m&daily=temperature_2m_max,temperature_2m_min&timezone=auto`);
      const fc = await fcRes.json();
      // for-loop pushes into days array (mutation)
      const days = [];
      for (let i = 0; i < 5; i++) {
        days.push({
          day: fc.daily.time[i],
          max: fc.daily.temperature_2m_max[i],
          min: fc.daily.temperature_2m_min[i],
        });
      }
      setData({ location: hit, current: fc.current, days });
    } catch (e: any) {
      setError(e.message);
    }
    setLoading(false);
  };

  const codeLabel = (code: number) => {
    if (code === 0) return "Clear";
    if (code <= 3) return "Cloudy";
    if (code <= 67) return "Rainy";
    return "Stormy";
  };

  return (
    <div>
      <input value={city} onChange={(e) => setCity(e.target.value)} />
      <button onClick={search}>Search</button>
      {loading && <p>loading…</p>}
      {error && <p>{error}</p>}
      {data && (
        <>
          <div>{data.location.name}: {Math.round(data.current.temperature_2m)}° {codeLabel(data.current.weather_code)}</div>
          {data.days.map((d: any) => (
            <div key={d.day}>{Math.round(d.max)}° / {Math.round(d.min)}°</div>
          ))}
        </>
      )}
    </div>
  );
}
Single file. `any` everywhere. Raw fetch. `for` loop with push (mutation). Magic numbers (0, 3, 67). No error class. No Zod. The reviewer can't tell where one concern ends and another begins.

import { ForecastSchema, GeoSchema } from "./schemas";
export class NetworkError extends Error { name = "NetworkError"; }
export class NotFoundError extends Error { name = "NotFoundError"; }
export class ApiError extends Error {
  constructor(public status: number, msg: string) { super(msg); }
}
export async function geocode(city: string) {
  const res = await fetch(...);
  if (!res.ok) throw new ApiError(res.status, "Geocode failed");
  const data = GeoSchema.parse(await res.json());
  if (data.results.length === 0) throw new NotFoundError(city);
  return data.results[0];
}

What changed (and what didn't)

  • Behavior: identical. Same URL, same UI, same Lighthouse score.
  • Files: 1 → 5 (page, apiClient, schemas, useForecast, plus 4 components).
  • Error handling: typed Error subclasses, branched with instanceof.
  • Validation: Zod schemas catch API shape changes at the boundary.
  • Mutation: the for/push loop became .map.
  • State: discriminated union { kind: "idle" | "loading" | "error" | "success" } instead of three separate booleans.
