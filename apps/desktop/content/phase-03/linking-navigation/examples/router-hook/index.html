Browser history (what the back button knows)

"use client";
import { useRouter, usePathname } from "next/navigation";

const router = useRouter();
const pathname = usePathname();

router.push("/dashboard");       // add to history, navigate
router.replace("/login");        // navigate WITHOUT adding to history
router.back();                    // go back, same as browser button
router.refresh();                 // re-fetch current route's data
