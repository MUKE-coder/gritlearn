Same "create note" mutation, two architectures

❌ Route handler + client fetch

2 files~45 linesJS required to work
app/api/notes/route.ts
import { z } from "zod";
import { db } from "@/lib/db";

const Schema = z.object({
  title: z.string().min(1),
  body: z.string().optional().default(""),
});

export async function POST(req: Request) {
  const json = await req.json();
  const parsed = Schema.safeParse(json);
  if (!parsed.success) {
    return Response.json(
      { error: parsed.error.issues[0].message },
      { status: 400 }
    );
  }
  const note = await db.note.create({ data: parsed.data });
  return Response.json(note, { status: 201 });
}
components/NoteForm.tsx
"use client";
import { useState } from "react";
import { useRouter } from "next/navigation";

export function NoteForm() {
  const router = useRouter();
  const [pending, setPending] = useState(false);
  const [error, setError] = useState<string | null>(null);

  async function onSubmit(e: React.FormEvent<HTMLFormElement>) {
    e.preventDefault();
    setPending(true);
    setError(null);
    const form = e.currentTarget;
    const data = Object.fromEntries(new FormData(form));
    const res = await fetch("/api/notes", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify(data),
    });
    setPending(false);
    if (!res.ok) {
      const { error } = await res.json();
      setError(error);
      return;
    }
    form.reset();
    router.refresh();
  }

  return (
    <form onSubmit={onSubmit}>
      <input name="title" />
      <textarea name="body" />
      {error && <p>{error}</p>}
      <button disabled={pending}>
        {pending ? "Saving…" : "Create"}
      </button>
    </form>
  );
}
Two files. JSON serialization on both ends. Manual pending state, manual error state, manual form.reset, manual router.refresh. Doesn't work without JS.

✓ Server action

1 file (+ small client component)~25 linesWorks without JS
app/notes/new/page.tsx (server component)
import { z } from "zod";
import { revalidatePath } from "next/cache";
import { db } from "@/lib/db";
import { SubmitBtn } from "@/components/SubmitBtn";

const Schema = z.object({
  title: z.string().min(1),
  body:  z.string().optional().default(""),
});

async function createNote(formData: FormData) {
  "use server";
  const parsed = Schema.parse({
    title: formData.get("title"),
    body:  formData.get("body"),
  });
  await db.note.create({ data: parsed });
  revalidatePath("/notes");
}

export default function NewNotePage() {
  return (
    <form action={createNote}>
      <input name="title" />
      <textarea name="body" />
      <SubmitBtn />
    </form>
  );
}
components/SubmitBtn.tsx (tiny client component)
"use client";
import { useFormStatus } from "react-dom";

export function SubmitBtn() {
  const { pending } = useFormStatus();
  return <button disabled={pending}>
    {pending ? "Saving…" : "Create"}
  </button>;
}
One real file (the page). One tiny helper. Validation, mutation, revalidation, pending state — all wired up by being in the right place. Works without JS.

The savings

  • ~45 lines → ~25 lines. Half the code for the same UX.
  • 2 files → 1 page + 1 tiny helper. No flipping between page.tsx and route.ts.
  • 0 fetch calls. The function looks local; runs on the server.
  • 0 useState lines. useFormStatus and the redirect handle the lifecycle.
  • Works without JavaScript. Progressive enhancement, free.
