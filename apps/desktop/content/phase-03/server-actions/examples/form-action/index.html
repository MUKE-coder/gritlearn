A complete form with server action

Three files. Real-world shape: validation, pending state, error display, list revalidation.

1 The server action

app/actions/notes.ts
"use server";

import { z } from "zod";
import { revalidatePath } from "next/cache";
import { db } from "@/lib/db";

const Schema = z.object({
  title: z.string().min(1, "Title required"),
  body:  z.string().optional().default(""),
});

type FormState = { error?: string };

export async function createNote(
  prev: FormState,
  formData: FormData
): Promise<FormState> {
  const parsed = Schema.safeParse({
    title: formData.get("title"),
    body:  formData.get("body"),
  });

  if (!parsed.success) {
    return { error: parsed.error.issues[0].message };
  }

  await db.note.create({ data: parsed.data });
  revalidatePath("/notes");
  return {}; // success — no error
}
The action takes `prev` (the previous form state) and `formData`. Returns the new state. Zod validates; revalidatePath refreshes the cached list.

2 The form (client component)

components/NoteForm.tsx
"use client";

import { useFormState, useFormStatus } from "react-dom";
import { createNote } from "@/app/actions/notes";

function SubmitButton() {
  const { pending } = useFormStatus();
  return (
    <button type="submit" disabled={pending}>
      {pending ? "Saving…" : "Create note"}
    </button>
  );
}

export function NoteForm() {
  const [state, formAction] = useFormState(createNote, {});

  return (
    <form action={formAction} className="space-y-3">
      <input name="title" placeholder="Title" className="input" />
      <textarea name="body" placeholder="Body" className="textarea" />
      {state.error && (
        <p className="text-red-500">{state.error}</p>
      )}
      <SubmitButton />
    </form>
  );
}
`useFormState` wraps the action and renders the returned state. `useFormStatus` (inside SubmitButton) reads the pending state of the enclosing form. No useState needed.

3 The page

app/notes/page.tsx
// server component — lists notes, renders the form
import { NoteForm } from "@/components/NoteForm";
import { db } from "@/lib/db";

export default async function NotesPage() {
  const notes = await db.note.findMany({
    orderBy: { createdAt: "desc" },
  });

  return (
    <main className="p-8 max-w-2xl">
      <h1 className="text-2xl font-bold">Notes</h1>
      <NoteForm />
      <ul className="mt-6 space-y-2">
        {notes.map((n) => (
          <li key={n.id}>{n.title}</li>
        ))}
      </ul>
    </main>
  );
}

What you don't write

  • No fetch() call.
  • No useState for the pending or error state.
  • No JSON serialization or parsing.
  • No API endpoint or route.ts.
  • No onSubmit handler with manual preventDefault.
  • No "refresh the list after success" logic — revalidatePath handles it.

The mutation, the validation, the pending state, the error display, the cache invalidation — all wired up by being where the framework expects them. The shape is what makes it work.
