function category(age) {
  // Guard clause — fail fast on invalid input
  if (typeof age !== "number" || Number.isNaN(age)) {
    return { label: "Enter a number", klass: "err" };
  }
  if (age < 0) {
    return { label: "Age can't be negative", klass: "err" };
  }

  // The ladder — first match wins
  if (age < 13) return { label: "Child",  klass: "child" };
  if (age < 20) return { label: "Teen",   klass: "teen" };
  if (age < 65) return { label: "Adult",  klass: "adult" };
  return            { label: "Senior", klass: "senior" };
}
