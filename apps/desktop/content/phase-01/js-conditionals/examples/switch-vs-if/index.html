Same logic, two ways

Both panes label a status string with one of four colors. Pick any status — both panes return the same label.

if / else if / else

function label(s) {
  if (s === "ok")       return "ok";
  if (s === "pending")  return "pending";
  if (s === "failed")   return "failed";
  return "unknown";
}

switch

function label(s) {
  switch (s) {
    case "ok":      return "ok";
    case "pending": return "pending";
    case "failed":  return "failed";
    default:        return "unknown";
  }
}
