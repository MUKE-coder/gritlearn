Typed Button

The component below is the everyday React + TypeScript pattern. The type up top is the contract; the call sites at the bottom of the script exercise it.

type ButtonProps = {
  label:    string;
  onClick:  () => void;
  variant?: "primary" | "ghost";    // optional, two legal values
  disabled?: boolean;
};

function Button({ label, onClick, variant = "primary", disabled }: ButtonProps) {
  return (
    <button className={"btn " + variant} onClick={onClick} disabled={disabled}>
      {label}
    </button>
  );
}
